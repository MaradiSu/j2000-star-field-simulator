cmake_minimum_required(VERSION 3.10)
project(j2000_star_field_simulator LANGUAGES CXX)
add_executable(star_sim star_sim.cpp)
set_target_properties(star_sim PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(star_sim PRIVATE /W4)
else()
  target_compile_options(star_sim PRIVATE -Wall -Wextra -pedantic)
endif()
enable_testing()
add_test(NAME geometry COMMAND star_sim --self-test)
find_package(PythonInterp 3 QUIET)
if(PYTHONINTERP_FOUND)
  add_test(NAME integration COMMAND "${PYTHON_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify.py" "$<TARGET_FILE:star_sim>")
endif()
